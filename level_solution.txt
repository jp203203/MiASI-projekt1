PROCEDURE move_items(n)
	REPEAT n
		TAKE
		ROTATE LEFT
		DROP
		ROTATE RIGHT
	ENDREPEAT
ENDPROC

move_items(2)

ENDWHILE

PROCEDURE if_item_move_else_walk()
	IF ITEM_TO RIGHT:
		ROTATE RIGHT
		MOVE 2
		ROTATE LEFT
		MOVE 2
	ELSE:
		MOVE 5
	ENDIF
ENDPROC

REPEAT 12 / 4
	if_item_move_else_walk()
ENDREPEAT

PROCEDURE move_avoid_trees()
	IF OBSTACLE_TO RIGHT
		ROTATE RIGHT
		MOVE 1
		ROTATE LEFT
		MOVE 3
		ROTATE LEFT
		MOVE 1
		ROTATE RIGHT
	ELSE
		MOVE 1
	ENDIF
ENDPROC

	move_avoid_trees()
ENDREPEAT
